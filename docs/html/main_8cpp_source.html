66template <typename T, size_t Capacity>
67void loadBuffer( // Параметры функции.
68 istringstream &line,
69 size_t count,
70 RingBuffer<T, Capacity> buffers[],
71 size_t &bufferCount) // Ссылка на переменную, копия не может так.
72{
73 if (count > Capacity)
74 {
75 throw runtime_error("Слишком много элементов в буфере");
76 }
77
78 if (bufferCount >= 10)
79 {
80 throw runtime_error("Превышено допустимое количество буферов");
81 }
82
83 RingBuffer<T, Capacity> buffer; // Промежуточный объект буфера, который будет заполнен и затем скопирован в массив.
84
85 for (size_t i = 0; i < count; i++)
86 {
87 T value;
88
89 if (!(line >> value))
90 {
91 throw runtime_error("Некорректные данные в файле");
92 }
93
94 buffer.push(value);
95 }
96
97 buffers[bufferCount] = buffer;
98 bufferCount++;
99}
